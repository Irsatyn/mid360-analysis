cmake_minimum_required(VERSION 3.16)
project(mid360_analysis)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp std_msgs sensor_msgs nav_msgs geometry_msgs visualization_msgs tf2 tf2_ros tf2_geometry_msgs)
foreach(dep IN LISTS dependencies)
  find_package(${dep} REQUIRED)
endforeach()
foreach(node imu_stats_node odom_analyzer_node)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} ${dependencies})
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/record.sh scripts/play.sh scripts/check_topics.sh DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_analysis test/test_analysis.cpp)
  target_include_directories(test_analysis PRIVATE include)
  foreach(test_name test_integration test_launch test_replay)
    ament_add_pytest_test(${test_name} test/${test_name}.py
      TIMEOUT 90 ENV MID360_NODE_DIR=${CMAKE_CURRENT_BINARY_DIR} ROS_LOG_DIR=${CMAKE_CURRENT_BINARY_DIR}/ros_logs)
  endforeach()
endif()
ament_export_include_directories(include)
ament_package()
